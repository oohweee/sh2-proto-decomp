# The DWARF has no typedefs, so anonymous types get layout-hash names. Each new name must be
# unique: it becomes both the struct tag and the typedef in types.h.

# layout is the game's own: from its DWARF (member names and offsets) where the type has them,
# otherwise from how the game uses it.
